1. Configurar Supabase (base de datos)

Supabase es PostgreSQL con API REST automática, auth y storage incluidos. Cree una cuenta en supabase.com y cree un nuevo proyecto.

Crear las tablas principales

Ejecute este SQL en el editor de Supabase (Database → SQL Editor):

SQL — Supabase SQL Editor 
-- Tabla de clientes/patrones
CREATE TABLE clientes (
  id          UUID DEFAULT gen_random_uuid() PRIMARY KEY,
  empresa     TEXT NOT NULL,
  rfc         TEXT UNIQUE NOT NULL,
  imss        TEXT,
  giro        TEXT,
  trabajadores INTEGER DEFAULT 0,
  contacto    TEXT,
  email       TEXT,
  telefono    TEXT,
  plan        TEXT DEFAULT 'Estándar',
  cuota       INTEGER DEFAULT 1799,
  semaforo    TEXT DEFAULT 'amarillo',
  score       INTEGER DEFAULT 50,
  stripe_customer_id TEXT,
  stripe_subscription_id TEXT,
  suscripcion_activa BOOLEAN DEFAULT true,
  created_at  TIMESTAMPTZ DEFAULT now(),
  updated_at  TIMESTAMPTZ DEFAULT now()
);

-- Tabla de documentos generados
CREATE TABLE documentos (
  id          UUID DEFAULT gen_random_uuid() PRIMARY KEY,
  cliente_id  UUID REFERENCES clientes(id) ON DELETE CASCADE,
  tipo        TEXT NOT NULL,
  nombre      TEXT NOT NULL,
  status      TEXT DEFAULT 'vigente',
  url_pdf     TEXT,
  metadata    JSONB,
  created_at  TIMESTAMPTZ DEFAULT now(),
  vence_at    TIMESTAMPTZ
);

-- Tabla de alertas
CREATE TABLE alertas (
  id          UUID DEFAULT gen_random_uuid() PRIMARY KEY,
  cliente_id  UUID REFERENCES clientes(id) ON DELETE CASCADE,
  tipo        TEXT NOT NULL,
  titulo      TEXT NOT NULL,
  descripcion TEXT,
  dias_restantes INTEGER,
  resuelta    BOOLEAN DEFAULT false,
  created_at  TIMESTAMPTZ DEFAULT now()
);

-- Tabla de solicitudes de asesoría
CREATE TABLE solicitudes (
  id          UUID DEFAULT gen_random_uuid() PRIMARY KEY,
  cliente_id  UUID REFERENCES clientes(id) ON DELETE CASCADE,
  tipo        TEXT,
  prioridad   TEXT DEFAULT 'normal',
  descripcion TEXT NOT NULL,
  trabajador  TEXT,
  salario     NUMERIC,
  status      TEXT DEFAULT 'pendiente',
  respuesta   TEXT,
  created_at  TIMESTAMPTZ DEFAULT now()
);

-- Tabla de pagos / historial de cobros
CREATE TABLE pagos (
  id          UUID DEFAULT gen_random_uuid() PRIMARY KEY,
  cliente_id  UUID REFERENCES clientes(id) ON DELETE CASCADE,
  stripe_invoice_id TEXT,
  monto       INTEGER,
  status      TEXT DEFAULT 'pagado',
  periodo     TEXT,
  created_at  TIMESTAMPTZ DEFAULT now()
);

-- Tabla de compliance por área
CREATE TABLE compliance (
  id          UUID DEFAULT gen_random_uuid() PRIMARY KEY,
  cliente_id  UUID REFERENCES clientes(id) ON DELETE CASCADE,
  area        TEXT NOT NULL,
  status      TEXT DEFAULT 'pendiente',
  notas       TEXT,
  updated_at  TIMESTAMPTZ DEFAULT now()
);

-- Índices para performance
CREATE INDEX idx_clientes_rfc ON clientes(rfc);
CREATE INDEX idx_documentos_cliente ON documentos(cliente_id);
CREATE INDEX idx_alertas_cliente ON alertas(cliente_id, resuelta);
CREATE INDEX idx_solicitudes_cliente ON solicitudes(cliente_id);

Row Level Security (RLS) — Seguridad por cliente

RLS garantiza que cada cliente sólo pueda ver sus propios datos, incluso si comparten la misma base de datos:

SQL — Row Level Security 
-- Activar RLS en todas las tablas de cliente
ALTER TABLE documentos ENABLE ROW LEVEL SECURITY;
ALTER TABLE alertas ENABLE ROW LEVEL SECURITY;
ALTER TABLE solicitudes ENABLE ROW LEVEL SECURITY;
ALTER TABLE pagos ENABLE ROW LEVEL SECURITY;
ALTER TABLE compliance ENABLE ROW LEVEL SECURITY;

-- Política: cliente solo ve SUS documentos
CREATE POLICY "cliente_documentos" ON documentos
  USING (cliente_id = (
    SELECT id FROM clientes
    WHERE rfc = auth.jwt()->>>'rfc'
  ));

-- Política: cliente solo ve SUS alertas
CREATE POLICY "cliente_alertas" ON alertas
  USING (cliente_id = (
    SELECT id FROM clientes
    WHERE rfc = auth.jwt()->>>'rfc'
  ));

-- El abogado (admin) ve TODO — identificado por rol
CREATE POLICY "admin_ve_todo" ON clientes
  USING (auth.jwt()->>>'role' = 'admin');

Autenticación con RFC

Configure Supabase Auth para que los clientes inicien sesión con email (RFC@dominio.clm) y contraseña. Así cada sesión tiene el RFC en el JWT y RLS se aplica automáticamente:

JavaScript — Supabase Client 
// En portal-cliente.html — reemplazar loginSubmit()
import { createClient } from '@supabase/supabase-js';

const supabase = createClient(
  'https://TU_PROJECT_ID.supabase.co',
  'TU_ANON_KEY'
);

async function loginConRFC(rfc, password) {
  // El email se construye como RFC@clicklaboral.mx internamente
  const { data, error } = await supabase.auth.signInWithPassword({
    email: `${rfc.toLowerCase()}@clicklaboral.mx`,
    password,
  });
  if (error) throw error;

  // Obtener datos del cliente
  const { data: cliente } = await supabase
    .from('clientes')
    .select('*')
    .eq('rfc', rfc)
    .single();

  return cliente;
}

async function obtenerDocumentos(clienteId) {
  const { data } = await supabase
    .from('documentos')
    .select('*')
    .eq('cliente_id', clienteId)
    .order('created_at', { ascending: false });
  return data;
}

async function logout() {
  await supabase.auth.signOut();
}

Configurar Webhook

El webhook escucha eventos de Stripe (cancelaciones, pagos exitosos o fallidos) y mantiene sincronizada la tabla clientes en Supabase. En Stripe Dashboard → Desarrolladores → Webhooks → Agregar endpoint, registre https://clicklaboral.mx/api/stripe-webhook escuchando los eventos customer.subscription.deleted, customer.subscription.updated, invoice.payment_succeeded e invoice.payment_failed. Stripe le dará un "Signing secret" (whsec_...) que va en la variable de entorno STRIPE_WEBHOOK_SECRET.

JavaScript — netlify/functions/stripe-webhook.js 
const Stripe = require('stripe');
const { createClient } = require('@supabase/supabase-js');

exports.handler = async (event) => {
  if (event.httpMethod !== 'POST') return { statusCode: 405, body: 'Método no permitido' };

  const stripe = Stripe(process.env.STRIPE_SECRET_KEY);
  const sig = event.headers['stripe-signature'];
  const rawBody = event.isBase64Encoded ? Buffer.from(event.body, 'base64') : event.body;

  let stripeEvent;
  try {
    stripeEvent = stripe.webhooks.constructEvent(rawBody, sig, process.env.STRIPE_WEBHOOK_SECRET);
  } catch (err) {
    return { statusCode: 400, body: `Webhook Error: ${err.message}` };
  }

  const supabase = createClient(process.env.SUPABASE_URL, process.env.SUPABASE_SERVICE_KEY);

  switch (stripeEvent.type) {
    case 'customer.subscription.deleted': {
      const sub = stripeEvent.data.object;
      await supabase.from('clientes').update({
        suscripcion_activa: false,
        semaforo: 'rojo',
      }).eq('stripe_customer_id', sub.customer);
      break;
    }
    case 'invoice.payment_succeeded': {
      const inv = stripeEvent.data.object;
      const { data: cliente } = await supabase
        .from('clientes').select('id')
        .eq('stripe_customer_id', inv.customer).single();
      if (cliente) {
        await supabase.from('pagos').insert({
          cliente_id: cliente.id,
          stripe_invoice_id: inv.id,
          monto: inv.amount_paid / 100,
          status: 'pagado',
          periodo: new Date(inv.period_start * 1000).toISOString(),
        });
      }
      break;
    }
  }

  return { statusCode: 200, body: 'ok' };
};

Versión completa (incluye también customer.subscription.updated e invoice.payment_failed) en netlify/functions/stripe-webhook.js, ya incluido en el ZIP del proyecto.

3. API Backend (Netlify Functions)

Todo vive en el mismo sitio de Netlify — no se necesita una cuenta ni un deploy separado en Vercel. La función ya está escrita en netlify/functions/crear-suscripcion.js, incluida en el ZIP del proyecto, con sus dependencias (stripe y @supabase/supabase-js) ya instaladas en su propio node_modules para que Netlify las empaquete correctamente incluso en un deploy manual (drag and drop).

Endpoint: crear suscripción desde checkout

JavaScript — netlify/functions/crear-suscripcion.js (resumen) 
const Stripe = require('stripe');
const { createClient } = require('@supabase/supabase-js');

// Mismo servicio completo en los 3 planes — solo cambia el precio
const PRICE_IDS = {
  basico:   'price_FALTA_BASICO_499',   // $499/mes — 1-8 trabajadores
  estandar: 'price_1Tn1EmBEbjDXzIvUr1Y3B1H8', // $1,799/mes — 9-49 trabajadores ✅
  pro:      'price_FALTA_PRO_7500',      // $7,500/mes — 50+ trabajadores
};

exports.handler = async (event) => {
  if (event.httpMethod !== 'POST')
    return { statusCode: 405, body: 'Método no permitido' };

  const stripe = Stripe(process.env.STRIPE_SECRET_KEY);
  const { paymentMethodId, plan, empresa, rfc, email, contacto } = JSON.parse(event.body);
  const priceId = PRICE_IDS[plan]; // plan = 'basico' | 'estandar' | 'pro'

  try {
    // 1. Crear customer en Stripe (con la tarjeta ya tokenizada por el navegador)
    const customer = await stripe.customers.create({
      email, name: empresa,
      payment_method: paymentMethodId,
      invoice_settings: { default_payment_method: paymentMethodId },
      metadata: { rfc, plan, contacto },
    });

    // 2. Crear suscripción al plan seleccionado
    const subscription = await stripe.subscriptions.create({
      customer: customer.id,
      items: [{ price: priceId }],
      expand: ['latest_invoice.payment_intent'],
      metadata: { rfc, empresa, plan },
    });

    // 3. Guardar cliente en Supabase + crear usuario del portal (no bloquea el cobro si falla)
    const supabase = createClient(process.env.SUPABASE_URL, process.env.SUPABASE_SERVICE_KEY);
    await supabase.from('clientes').insert({
      empresa, rfc, email, contacto, plan: NOMBRES_PLAN[plan], cuota: CUOTAS_PLAN[plan],
      stripe_customer_id: customer.id, stripe_subscription_id: subscription.id,
      suscripcion_activa: true,
    });

    return { statusCode: 200, body: JSON.stringify({ success: true, subscriptionId: subscription.id }) };
  } catch (err) {
    return { statusCode: 402, body: JSON.stringify({ error: err.message }) };
  }
};

Esta es una versión resumida para lectura rápida. El archivo real incluye manejo de 3D Secure, validaciones, CORS y mensajes de error en español — está completo en netlify/functions/crear-suscripcion.js.

4. Variables de entorno y despliegue

1
Subir el ZIP con la función incluida
Use el ZIP completo del proyecto (incluye netlify.toml y la carpeta netlify/functions/ con sus dependencias ya instaladas) y arrástrelo al deploy zone de Netlify, como ya lo ha hecho con las versiones anteriores.
2
Configurar variables de entorno en Netlify
En su sitio de Netlify → Site configuration → Environment variables → Add a variable. Agregue las claves del siguiente bloque.
3
Volver a desplegar
Después de agregar o cambiar variables de entorno, vuelva a subir el ZIP del proyecto (drag and drop) para que la función las tome en cuenta — los cambios de variables de entorno no aplican a deploys ya existentes.
4
Probar con una tarjeta de prueba
Use la tarjeta de prueba 4242 4242 4242 4242 con cualquier fecha futura y CVC en checkout.html mientras su clave de Stripe esté en modo de prueba (sk_test_... / pk_test_...). Cuando confirme que todo funciona, cambie a sus claves sk_live_... / pk_live_....

Variables de entorno requeridas

Netlify → Site configuration → Environment variables 
# Stripe — obligatoria para cobrar

⚠️ Nunca pegues claves secretas en el código ni se las mandes a nadie por chat

La STRIPE_SECRET_KEY y la SUPABASE_SERVICE_KEY dan acceso total — solo van en el panel de variables de entorno de Netlify, nunca en un archivo HTML/JS que se sube al repositorio o al ZIP. La STRIPE_PUBLISHABLE_KEY (la que sí va en checkout.html) es la única que es segura de exponer en el navegador.

💡 ¿Por qué incluir node_modules en el ZIP?

Un deploy manual (drag and drop) en Netlify no ejecuta npm install — solo sube los archivos tal cual. Por eso las dependencias ya vienen instaladas dentro de netlify/functions/node_modules/ en el ZIP del proyecto; así Netlify puede empaquetar la función sin necesitar un paso de build. Si en el futuro conecta el sitio a un repositorio de Git, esto deja de ser necesario.
