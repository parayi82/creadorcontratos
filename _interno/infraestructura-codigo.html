Proxy seguro para la API de Anthropic — la API key nunca se expone al frontend
Edge Function 
/**
 * /api/claude-proxy.js
 * Proxy seguro para Claude API — mantiene la API key en el servidor.
 * El frontend llama a /api/claude en lugar de api.anthropic.com directamente.
 * Incluye: autenticación de sesión, rate limiting y logging.
 */

import { createClient } from '@supabase/supabase-js';

const ALLOWED_ORIGINS = [
  'https://compliance.mx',
  'https://www.compliance.mx',
  'http://localhost:3000', // desarrollo local
];

const SYSTEM_PROMPTS = {
  asesor: `Eres un abogado laboralista mexicano experto. 
Cita siempre los artículos de la LFT. Responde en español formal.
Máximo 400 palabras por respuesta en el portal de cliente.`,
  clausulas: `Eres un redactor jurídico especializado en contratos laborales mexicanos.
Redacta cláusulas en formato: CLÁUSULA N. — NOMBRE seguido del texto y fundamento legal.`,
  stps: `Eres un experto en trámites ante STPS, IMSS y Tribunales Laborales.
Redacta escritos formales con: encabezado, hechos, fundamento de derecho, petitorios.`,
  liquidacion: `Eres experto en cálculo de prestaciones laborales conforme a la LFT.
Calcula paso a paso, muestra cada componente con su artículo fundamento.`,
  dictamen: `Eres consultor de compliance laboral. Elabora dictámenes con semáforo 
de riesgos (🔴🟡🟢) y plan de acción priorizado con artículos de la LFT.`,
};

export default async function handler(req) {
  // CORS
  const origin = req.headers.get('origin');
  const corsHeaders = {
    'Access-Control-Allow-Origin': ALLOWED_ORIGINS.includes(origin) ? origin : '',
    'Access-Control-Allow-Methods': 'POST, OPTIONS',
    'Access-Control-Allow-Headers': 'Content-Type, Authorization',
  };
  if (req.method === 'OPTIONS') return new Response('', { headers: corsHeaders });
  if (req.method !== 'POST') return new Response('Method not allowed', { status: 405 });

  try {
    const { messages, herramienta = 'asesor', contextoEmpresa, rfc } = await req.json();

    // Verificar que el cliente tiene suscripción activa
    if (rfc) {
      const supabase = createClient(
        process.env.SUPABASE_URL,
        process.env.SUPABASE_SERVICE_KEY
      );
      const { data: cliente } = await supabase
        .from('clientes')
        .select('suscripcion_activa, plan')
        .eq('rfc', rfc)
        .single();
      if (!cliente?.suscripcion_activa) {
        return new Response(
          JSON.stringify({ error: 'Suscripción inactiva' }),
          { status: 403, headers: corsHeaders }
        );
      }
    }

    // Construir system prompt con contexto de empresa
    let systemPrompt = SYSTEM_PROMPTS[herramienta] || SYSTEM_PROMPTS.asesor;
    if (contextoEmpresa) {
      systemPrompt += `\n\nCONTEXTO DEL CLIENTE: ${JSON.stringify(contextoEmpresa)}`;
    }

    // Llamar a Claude API
    const response = await fetch('https://api.anthropic.com/v1/messages', {
      method: 'POST',
      headers: {
        'Content-Type': 'application/json',
        'x-api-key': process.env.ANTHROPIC_API_KEY,
        'anthropic-version': '2023-06-01',
      },
      body: JSON.stringify({
        model: 'claude-sonnet-4-6',
        max_tokens: 1000,
        system: systemPrompt,
        messages: messages.slice(-12), // últimos 12 mensajes = contexto razonable
      }),
    });

    if (!response.ok) {
      const err = await response.json();
      throw new Error(err.error?.message || `Anthropic error ${response.status}`);
    }

    const data = await response.json();
    return new Response(
      JSON.stringify({ texto: data.content?.[0]?.text || '' }),
      { status: 200, headers: { 'Content-Type': 'application/json', ...corsHeaders } }
    );

  } catch (err) {
    console.error('Claude proxy error:', err);
    return new Response(
      JSON.stringify({ error: err.message }),
      { status: 500, headers: corsHeaders }
    );
  }
}

/**
 * /api/procesar-alertas.js
 * Se ejecuta cada día hábil a las 9am UTC-6 (México).
 * Revisa todos los clientes activos, calcula días a vencimiento
 * y envía emails cuando quedan 30, 15 o 7 días.
 * Configurar en vercel.json: "crons": [{"path":"/api/procesar-alertas","schedule":"0 15 * * 1-5"}]
 */

import { createClient } from '@supabase/supabase-js';
import { Resend } from 'resend';

const supabase = createClient(process.env.SUPABASE_URL, process.env.SUPABASE_SERVICE_KEY);
const resend = new Resend(process.env.RESEND_API_KEY);

// Obligaciones periódicas que hay que rastrear por cliente
const calcularVencimientos = (cliente) => {
  const hoy = new Date();
  const vencimientos = [];

  // Aguinaldo — vence el 20 de diciembre
  const aguinaldo = new Date(hoy.getFullYear(), 11, 20);
  const diasAguinaldo = Math.ceil((aguinaldo - hoy) / (1000*60*60*24));
  if (diasAguinaldo > 0 && diasAguinaldo <= 30) {
    vencimientos.push({ tipo: 'urgent', titulo: 'Pago de aguinaldo', dias: diasAguinaldo, art: 'Art. 87 LFT' });
  }

  // PTU — vence 60 días después de la declaración anual
  const ptu = new Date(hoy.getFullYear(), 4, 30); // aprox. 30 mayo
  const diasPTU = Math.ceil((ptu - hoy) / (1000*60*60*24));
  if (diasPTU > 0 && diasPTU <= 30) {
    vencimientos.push({ tipo: 'warn', titulo: 'Pago de PTU', dias: diasPTU, art: 'Art. 122 LFT' });
  }

  // NOM-035 — anual desde la fecha de evaluación previa
  if (cliente.venc_nom035) {
    const vencNOM = new Date(cliente.venc_nom035);
    const diasNOM = Math.ceil((vencNOM - hoy) / (1000*60*60*24));
    if (diasNOM > 0 && diasNOM <= 30) {
      vencimientos.push({ tipo: diasNOM<=7?'urgent':'warn', titulo: 'Evaluación NOM-035-STPS', dias: diasNOM, art: 'NOM-035-STPS-2018' });
    }
  }

  // Reglamento interior — cada 2 años desde el depósito
  if (cliente.venc_reglamento) {
    const vencReg = new Date(cliente.venc_reglamento);
    const diasReg = Math.ceil((vencReg - hoy) / (1000*60*60*24));
    if (diasReg > 0 && diasReg <= 30) {
      vencimientos.push({ tipo: 'warn', titulo: 'Renovación Reglamento Interior', dias: diasReg, art: 'Art. 424 LFT' });
    }
  }

  return vencimientos;
};

export default async function handler(req) {
  // Verificar que es llamada del cron de Vercel
  const authHeader = req.headers.get('authorization');
  if (authHeader !== `Bearer ${process.env.CRON_SECRET}`) {
    return new Response('Unauthorized', { status: 401 });
  }

  // Obtener todos los clientes activos
  const { data: clientes } = await supabase
    .from('clientes')
    .select('*')
    .eq('suscripcion_activa', true);

  let alertasEnviadas = 0;

  for (const cliente of clientes) {
    const vencimientos = calcularVencimientos(cliente);
    if (!vencimientos.length) continue;

    // Guardar alertas en Supabase
    for (const v of vencimientos) {
      await supabase.from('alertas').upsert({
        cliente_id: cliente.id,
        tipo: v.tipo,
        titulo: v.titulo,
        descripcion: `Fundamento: ${v.art} — Quedan ${v.dias} días`,
        dias_restantes: v.dias,
      }, { onConflict: 'cliente_id,titulo' });
    }

    // Solo enviar email si alguno es urgente (≤7 días) o exactamente en 30/15 días
    const enviarEmail = vencimientos.some(v =>
      v.dias <= 7 || v.dias === 15 || v.dias === 30
    );

    if (enviarEmail && cliente.email) {
      const urgentes = vencimientos.filter(v => v.dias <= 7);
      const asunto = urgentes.length
        ? `🚨 Urgente: ${urgentes[0].titulo} vence en ${urgentes[0].dias} días`
        : `⚠️ Recordatorio de compliance — ${vencimientos.length} obligaciones próximas`;

      await resend.emails.send({
        from: 'ClickLaboral.mx <alertas@compliance.mx>',
        to: cliente.email,
        subject: asunto,
        html: generarEmailAlerta(cliente, vencimientos),
      });
      alertasEnviadas++;
    }
  }

  return new Response(
    JSON.stringify({ ok: true, clientes: clientes.length, alertasEnviadas }),
    { status: 200 }
  );
}

api/cancelar-suscripcion.js
Cancela la suscripción en Stripe y actualiza el estado en Supabase
Destructivo 
import Stripe from 'stripe';
import { createClient } from '@supabase/supabase-js';

const stripe = new Stripe(process.env.STRIPE_SECRET_KEY);
const supabase = createClient(process.env.SUPABASE_URL, process.env.SUPABASE_SERVICE_KEY);

export default async function handler(req) {
  if (req.method !== 'POST') return new Response('Method not allowed', { status: 405 });

  const { rfc, motivo, adminSecret } = await req.json();

  // Solo el admin puede cancelar
  if (adminSecret !== process.env.ADMIN_SECRET) {
    return new Response(JSON.stringify({ error: 'No autorizado' }), { status: 403 });
  }

  const { data: cliente } = await supabase
    .from('clientes')
    .select('stripe_subscription_id, empresa, email')
    .eq('rfc', rfc)
    .single();

  if (!cliente) return new Response(JSON.stringify({ error: 'Cliente no encontrado' }), { status: 404 });

  // Cancelar al final del período (no inmediatamente = buena práctica)
  await stripe.subscriptions.update(cliente.stripe_subscription_id, {
    cancel_at_period_end: true,
    metadata: { motivo_cancelacion: motivo },
  });

  await supabase.from('clientes').update({
    semaforo: 'amarillo',
  }).eq('rfc', rfc);

  return new Response(
    JSON.stringify({ ok: true, mensaje: `Suscripción de ${cliente.empresa} cancelada al final del período` }),
    { status: 200 }
  );
}

export const supabase = createClient(
  process.env.NEXT_PUBLIC_SUPABASE_URL || process.env.SUPABASE_URL,
  process.env.NEXT_PUBLIC_SUPABASE_ANON_KEY || process.env.SUPABASE_ANON_KEY
);

// Cliente con privilegios de servicio — SOLO en servidor
export const supabaseAdmin = createClient(
  process.env.SUPABASE_URL,
  process.env.SUPABASE_SERVICE_KEY,
  { auth: { autoRefreshToken: false, persistSession: false } }
);

// Helpers de consulta frecuente
export async function getCliente(rfc) {
  const { data, error } = await supabaseAdmin
    .from('clientes')
    .select('*')
    .eq('rfc', rfc.toUpperCase())
    .single();
  if (error) throw error;
  return data;
}

export async function getDocumentos(clienteId) {
  const { data } = await supabaseAdmin
    .from('documentos')
    .select('*')
    .eq('cliente_id', clienteId)
    .order('created_at', { ascending: false });
  return data || [];
}

export async function getAlertasActivas(clienteId) {
  const { data } = await supabaseAdmin
    .from('alertas')
    .select('*')
    .eq('cliente_id', clienteId)
    .eq('resuelta', false)
    .order('dias_restantes', { ascending: true });
  return data || [];
}

export async function getMRR() {
  const { data } = await supabaseAdmin
    .from('clientes')
    .select('cuota')
    .eq('suscripcion_activa', true);
  return (data || []).reduce((sum, c) => sum + c.cuota, 0);
}
